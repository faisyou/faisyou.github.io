Normal TLS vs Mutual TLS

Normal HTTPS

Genesys→Customer server
  1. Customer presents server certificate.
  2. Genesys validates server identity.
  3. Encrypted TLS session is established.
  4. HTTP request is sent.

Mutual TLS

Genesys↔Customer server
  1. Customer presents server certificate.
  2. Genesys validates server identity.
  3. Customer requests a client certificate.
  4. Genesys presents its client certificate.
  5. Customer validates client identity.
  6. TLS session is established; HTTP follows.
mTLS adds CLIENT CERTIFICATE AUTHENTICATION — it does not replace API keys, OAuth, or application authorization.

What this test proves — and what it does not

THIS TEST PROVES

  • AWS endpoint is reachable.
  • Genesys can establish TLS with it.
  • The server can observe whether a client cert is presented.
  • Presented certificate metadata can be captured.
  • Customer trust can be enforced.
  • The Data Action can complete an HTTP request.

THIS TEST DOES NOT AUTOMATICALLY PROVE

  • Production certificate lifecycle and rotation are correct.
  • High availability or production gateway behavior.
  • Every possible region/path.
  • Application authorization policy.
  • Security of the entire production integration.
A diagnostic proves specific layers. It should not be used to overclaim production readiness.

Appendix A — Full Node.js server source

const https = require('https');
const fs = require('fs');

const PORT = Number(process.env.PORT || 443);
const KEY_PATH = process.env.SERVER_KEY || './certs/server-key.pem';
const CERT_PATH = process.env.SERVER_CERT || './certs/server-cert.pem';
const CA_PATH = process.env.CLIENT_CA || '';
const STRICT_CLIENT_AUTH = process.env.STRICT_CLIENT_AUTH === 'true';

function readOptional(path) {
  return path ? fs.readFileSync(path) : undefined;
}

const tlsOptions = {
  key: fs.readFileSync(KEY_PATH),
  cert: fs.readFileSync(CERT_PATH),
  requestCert: true,
  rejectUnauthorized: STRICT_CLIENT_AUTH,
  ca: readOptional(CA_PATH)
};

function certSummary(socket) {
  const cert = socket.getPeerCertificate(true);
  const presented = Boolean(cert && Object.keys(cert).length);
  return {
    clientCertificatePresented: presented,
    authorized: socket.authorized,
    authorizationError: socket.authorizationError || null,
    subject: presented ? cert.subject : null,
    issuer: presented ? cert.issuer : null,
    serialNumber: presented ? cert.serialNumber : null,
    validFrom: presented ? cert.valid_from : null,
    validTo: presented ? cert.valid_to : null,
    fingerprint256: presented ? cert.fingerprint256 : null
  };
}

const server = https.createServer(tlsOptions, (req, res) => {
  try {
    const summary = certSummary(req.socket);
    const result = {
      timestamp: new Date().toISOString(),
      method: req.method,
      path: req.url,
      remoteAddress: req.socket.remoteAddress,
      tlsProtocol: req.socket.getProtocol(),
      ...summary
    };

    console.log(JSON.stringify(result));

    if (req.url !== '/test') {
      res.writeHead(404, { 'content-type': 'application/json' });
      return res.end(JSON.stringify({ error: 'not_found' }));
    }

    res.writeHead(200, { 'content-type': 'application/json' });
    res.end(JSON.stringify(result, null, 2));
  } catch (err) {
    console.error('request_error', err);
    if (!res.headersSent) {
      res.writeHead(500, { 'content-type': 'application/json' });
    }
    res.end(JSON.stringify({ error: 'internal_error' }));
  }
});

server.on('tlsClientError', (err, socket) => {
  console.error('tls_client_error', {
    message: err.message,
    remoteAddress: socket && socket.remoteAddress
  });
});

server.on('error', (err) => {
  console.error('server_error', err);
  process.exitCode = 1;
});

server.listen(PORT, '0.0.0.0', () => {
  console.log(`mTLS diagnostic server listening on ${PORT}`);
  console.log(`strict client auth: ${STRICT_CLIENT_AUTH}`);
});
requestCert: true = ask the client for a certificaterejectUnauthorized = enforce validation resultca = trust material when strict mode is enabled
